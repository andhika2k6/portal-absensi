Skema Tabel Supabase & Bucket foto-absensi

Jalankan perintah SQL berikut pada SQL Editor Supabase Anda, serta buat Storage Bucket publik bernama foto-absensi:

-- 1. Tabel Master Panitia
create table if not exists public.panitia (
  id_panitia text primary key,
  nrp text unique not null,
  nama text not null,
  role text default 'panitia', -- 'admin' atau 'panitia'
  asal_sma text,
  zona_domisili text,
  bersedia_jauh text default 'Bersedia',
  tanggal_halangan text default '-',
  info_tambahan text
);

-- 2. Tabel Plotting & Bukti Absensi Geotag
create table if not exists public.plotting_absensi (
  id_plotting text primary key,
  id_sekolah text not null,
  nama_sma text not null,
  zona_sekolah text,
  kategori_lokasi text,
  tanggal_visit text,
  tim int,
  kursi int,
  id_panitia text,
  nrp text,
  nama_panitia text,
  is_alumni boolean default false,
  status_hadir text default 'Belum Hadir',
  foto_url text,
  latitude numeric,
  longitude numeric,
  akurasi_meter numeric,
  waktu_absen timestamptz
);
